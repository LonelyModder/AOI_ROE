latium_01_italy = { # Alessian Revolt
	icon = latium_01
	header = latium_01_italy #mission_image_phrygia
	
	repeatable = no

    chance = {
		factor = 10
	}
	
	ai_chance = { factor = 10 }

	potential = { 
		NOT = { has_variable = latium_01_italy }
		primary_culture = rroman
		region:modded_italia_region = {
            any_region_province = {
                owned_or_subject_owned = root
            }
        }
		tag=AAA
		is_monarchy = yes
		is_subject = no
		has_civil_war = no
		NOT = { has_global_variable = italy_fail }
	}
	
	abort = {	
		OR = {
			is_monarchy = no
			NOT = { primary_culture = rroman }
			custom_tooltip = {
				text = latium_01_italy_abort_tt
				has_global_variable = italy_fail
			}
			ai_mission_back_out_trigger = yes
		}
	}
	
	on_start = {
    	start_mission_ai_effect = yes
    	save_scope_as = me_italy_scope
		add_country_modifier = {
	        name = revolt_starts
	        duration = 3650 # 10 years
	    }
    }
	
	on_abort = {
    	custom_tooltip = ant_1_mission_cooldown_tt
    	if = {
	    	limit = { has_country_modifier = revolt_starts }
    		remove_country_modifier = revolt_starts
	    }
		set_variable = {
			name = latium_01_italy
			days = 7300
		}
    }
	
	on_completion = {
		add_country_modifier = {
    		name = revolt_victory
    		duration = 7300 # 20 years
    	}
    	complete_mission_effect = yes #
    	#trigger_event = { id = me_antigonid_1_vision.2 }
    }
	
	
	latium_01_italy_1 = { # Rising in Revolt
		requires = { latium_01_italy_12 }

		allow = {

			}

		}

		}
	}
