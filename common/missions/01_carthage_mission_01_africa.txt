	repeatable = no

		factor = 10
	}

		NOT = { has_variable = carthage_01_africa }

		tag=DAT
		is_monarchy = yes
		is_subject = no
		has_civil_war = no
		NOT = { has_global_variable = carthage_fail }

	    }
