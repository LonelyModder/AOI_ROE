	type = country_event
	title = "lydia_01_anatolia_1.15.t"
	desc = "lydia_01_anatolia_1.15.desc"
	fire_only_once = yes
	picture = greek_court
	
	trigger = {
		tag=GAC
	}
	
	option = { 
		name = "lydia_01_anatolia_1.15.a"
		c:GAY = {
			every_owned_province = {
				add_claim = c:GAC
			}
		}
		p:2649 = { #Treres Tribe
			while = {
                count = 65
                define_pop = {
					type = tribesmen
					culture = brygian
					religion = thracic_pantheon
                }
            } 
			create_country = {
				change_government = tribal_kingdom
				if = {
					limit = {
						NOT = {
							primary_culture = brygian
						}
					}
					set_primary_culture_cleanup_effect = {
						NEW_PRIMARY_CULTURE = brygian
						MAKE_OLD_CULTURE_INTEGRATED = yes
					}
				}
				set_country_religion = thracic_pantheon
				change_country_name = "TRERES_NAME"
				change_country_adjective = "TRERES_ADJECTIVE"
			}
			add_country_modifier = {
				name = lepontic_invasion_modifier
				duration = 14600
			}
		}
		p:2649.owner = {
			declare_war_with_wargoal = {
				war_goal = conquer_wargoal
				province = p:2652
				target = p:2652.owner
			}
		}
		set_global_variable = {
			name = trerian_brotherly_aid
		}
		trigger_event = { 
			id = lydia_01_anatolia_1.19
		}
	}
}
