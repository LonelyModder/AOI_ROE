cymmerian_invasion_ec.3 = {  
	type = country_event
	hidden = yes
	fire_only_once = yes
	
	trigger = {
		primary_culture = cymmeric
		war = no
		owns = 3268 #Turaka Tribe
		p:3268 = {
			is_capital = yes
		}
		is_ai = yes
		NOT = { has_global_variable = turaka_landed }
	}
	
	immediate = { 
		p:3268 = { #Turaka Tribe
			create_country = {
				change_government = despotic_monarchy #New Government?
				set_primary_culture = cymmeric
				set_country_religion = scythic_pantheon
				change_country_name = "AnatolianCymmeria_NAME"
				change_country_adjective = "AnatolianCymmeria_ADJECTIVE"
			}
		}
		every_owned_province = {
			set_conquered_by = p:3268.owner
		}
		set_global_variable = {
			name = turaka_landed
		}
	}
}
