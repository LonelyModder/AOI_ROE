macedon_01_macedonia_1.4 = {  
	type = country_event
	title = "macedon_01_macedonia_1.4.t"
	desc = "macedon_01_macedonia_1.4.desc"
	fire_only_once = yes
	picture = macedon_pic_1
	
	trigger = {
		tag=BAA
	}
	
	option = { 
		name = "macedon_01_macedonia_1.4.a"
		if = {
			limit = {
				NOT = {
					capital_scope = {
						province_id = 1295
					}
				}
			}
			set_capital = 1295
			hidden_effect = {
				p:1295.state = {
					set_state_capital = 1295
				}
			}
		}
		add_stability = 25
		p:1295 = {
			capital_formable_medium_effect = yes
			formable_capital_modifier_normal_effect = yes
		}
		p:1295 = { #Pella
			while = {
                count = 1
                define_pop = {
					type = nobles
					culture = gmacedonian
					religion = grecoid_pantheon
                }
            }
			while = {
                count = 4
                define_pop = {
					type = citizen
					culture = gmacedonian
					religion = grecoid_pantheon
                }
            }
			while = {
                count = 2
                define_pop = {
					type = freemen
					culture = gmacedonian
					religion = grecoid_pantheon
                }
            }
		}
		add_2_free_province_investments = yes
	}
}
