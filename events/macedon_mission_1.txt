macedon_01_macedonia_1.4 = {  
	type = country_event
	title = "macedon_01_macedonia_1.4.t"
	desc = "macedon_01_macedonia_1.4.desc"
	fire_only_once = yes
	picture = macedon_pic_1
	
	trigger = {
		tag=BAA
	}
	
	option = { 
		name = "macedon_01_macedonia_1.4.a"
		if = {
			limit = {
				NOT = {
					capital_scope = {
						province_id = 1295
					}
				}
			}
			set_capital = 1295
			hidden_effect = {
				p:1295.state = {
					set_state_capital = 1295
				}
			}
		}
		add_stability = 25
		p:1295 = {
			capital_formable_medium_effect = yes
			formable_capital_modifier_normal_effect = yes
		}
		p:1295 = { #Pella
			while = {
                count = 1
                define_pop = {
					type = nobles
					culture = gmacedonian
					religion = grecoid_pantheon
                }
            }
			while = {
                count = 4
                define_pop = {
					type = citizen
					culture = gmacedonian
					religion = grecoid_pantheon
                }
            }
			while = {
                count = 2
                define_pop = {
					type = freemen
					culture = gmacedonian
					religion = grecoid_pantheon
                }
            }
		}
		add_2_free_province_investments = yes
	}
}

		OR = {
			tag = TBH #Bisalta
			tag = TAN #Brygia
			owns = 1353 
			owns = 1345 
			owns = 1344 
			owns = 1354 
		}
		OR = {
			country_culture_group = thracianic
			country_culture_group = illyric
		}
		has_global_variable = macedonian_ultimatum
	}
	
	option = { 
		name = "macedon_01_macedonia_1.12.a"
		set_global_variable = {
			name = macedonian_ultimatum_accepted
		}
		if = {
			limit = {
				OR = {
					tag = TBH
					tag = TAN
				}
			}
			root = {
				make_subject = {
					type = feudatory
					target = c:BAA
				}
			}
		}
		else_if = {
			limit = {
				OR = {
					owns = 1353 
					owns = 1345 
					owns = 1344 
					owns = 1354 
				}
			}
			p:1353 = { set_conquered_by = c:BAA }
			p:1345 = { set_conquered_by = c:BAA }
			p:1344 = { set_conquered_by = c:BAA }
			p:1354 = { set_conquered_by = c:BAA }
		}
		ai_chance = { base = 20 }
	}
	
	option = { 
		name = "macedon_01_macedonia_1.12.b"
		set_global_variable = {
			name = macedonian_ultimatum_rejected
		}
		p:1353 ={
			every_area_province = {
				limit = {
					owner = root 
				}

			}
		}

	}
