kush_01_egypt_1.14 = {  
	type = country_event
	title = "kush_01_egypt_1.14.t"
	desc = "kush_01_egypt_1.14.desc"
	#fire_only_once = yes
	picture = kush_pic_1
	
	trigger = {
		tag = EAD
		has_global_variable = wadi_hammamat_colonization
		NOT = { has_global_variable = wadi_hammamat_colonized }
	}
	
	option = { 
		name = "kush_01_egypt_1.14.a"
		random_ownable_province = {
			limit = {
				has_owner = no
				OR = {
					is_in_area = berenice_area
					is_in_area = hammamat_area
					is_in_area = hormos_area
					is_in_area = redcoast_area
				}
				any_neighbor_province = {
                    count >= 2
                    has_owner = yes
                    owner = c:EAD
				}
			}
			set_conquered_by = c:EAD
			add_province_modifier = {
                name = colonization_progress
                duration = 365
            }
		}
		add_manpower = -0.25
		add_treasury = -5
		ai_chancee = { base = 80 }
	}
	
	option = { 
		name = "kush_01_egypt_1.14.b"
		ai_chancee = { base = 20 }
	}
}

kush_01_egypt_1.18 = {  
	type = country_event
	title = "kush_01_egypt_1.18.t"
	desc = "kush_01_egypt_1.18.desc"
	fire_only_once = yes
	picture = tefnakht_egypt_pic
	hidden = yes
	
	trigger = {
		tag = EAD
		owns = 2043
	}
	
	immediate = {
		
		create_character = {
			first_name = "Tefnakht"
			family_name = "Merriamun"
			no_stats = yes
			culture = lower_kemetic
			religion = kemetic_pantheon
			age = 23
			good_martial_effect = yes
			good_charisma_effect = yes
			average_finesse_effect = yes
			good_zeal_effect = yes
			add_gold=500
			save_scope_as = tefnakht_gov
		}
	}
	
	option = { 
		name = "kush_01_egypt_1.18.a"
		#Tefnakhts Uprising
		p:1968 = { #24th Dynasty
			create_country = {
				change_government = theocratic_monarchy
				set_primary_culture = lower_kemetic
				set_country_religion = kemetic_pantheon
				change_country_name = "Tefnakhts_NAME"
				change_country_adjective = "Tefnakhts_ADJECTIVE"
				set_as_ruler=scope:tefnakht_gov
			}
		}
		p:1968.owner = {
			add_country_modifier = {
				name = lepontic_invasion_modifier
				duration = 730
			}	
		}
		p:1966.owner = {
			declare_war_with_wargoal = {
				war_goal = conquer_wargoal
				province = p:1968
				target = p:1968.owner
			}
		}
		set_global_variable = {
			name = tefnakhts_rebelion
		}
	}
}
