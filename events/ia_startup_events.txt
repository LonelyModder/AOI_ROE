esfa_startup.1 = { #Italics
    type = country_event
    title = esfa_startup.1.t
    desc = esfa_startup.1.desc
    picture = welcoming_window_1
	
    trigger = {
		OR = {
			country_culture_group = latinic
			country_culture_group = italic
			country_culture_group = thyrrenic
		}
    }

    option = {    
        name = esfa_startup.1.a
    }
}

esfa_startup.2 = { #Celtic
    type = country_event
    title = esfa_startup.2.t
    desc = esfa_startup.2.desc
    picture = welcoming_window_1
	
    trigger = {
		country_culture_group = celtic
    }

    option = {    
        name = esfa_startup.2.a
		c:ABL = {
			add_country_modifier = {
				name = lepontic_invasion_modifier
				duration = 7300 # 20 years
			}
			#Claims in Po valley
			c:AAG = {
				every_owned_province = {
					limit = { is_in_region = modded_galliacisalpina_region }
					add_claim = c:ABL
				}
			}
		}
    }
}
