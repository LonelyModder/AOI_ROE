namespace = bloodlines_startup

bloodlines_startup.1 = { #Nobilitas

	type = country_event
	hidden = yes
	
	trigger = {
		tag = AAA
	}
	
	immediate = {
		every_character = {
			limit = {
				OR = {
					has_same_family = char:1
					has_same_family = char:2
					has_same_family = char:3
					has_same_family = char:4
				}
			}
			add_trait = nobilitas
		
		}
		hidden_effect = {
			trigger_event = { id = bloodlines_setup_completion.1 }
		}
	
	}

}
