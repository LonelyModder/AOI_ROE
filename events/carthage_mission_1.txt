carthage_01_africa_1.9 = {  
	type = country_event
	hidden = yes
	fire_only_once = yes
	
	trigger = {
		tag=DAT
	}
	
	immediate = { 
		set_global_variable = {
			name = Oea_colonisation_started
		}
		if = {
			limit = {
				p:3754.owner = {
					OR = {
						exists = no
						tag = DAT
					}
				}
			}
			trigger_event = { 
				id = carthage_01_africa_1.10 #Create Oea
				days = 30
			}
		}
		else_if = {
			limit = {
				p:3754.owner = {
					is_tribal = yes
				}
			}
			trigger_event = { 
				id = carthage_01_africa_1.11 #Negotiate Oea
				days = 30
			}
		}
		
	}
}

carthage_01_africa_1.10 = {  
	type = country_event
	title = "carthage_01_africa_1.10.t"
	desc = "carthage_01_africa_1.10.desc"
	fire_only_once = yes
	picture = carthage_pic_1
	
	trigger = {
		tag=DAT
	}
	
	option = { 
		name = "carthage_01_africa_1.10.a"
		add_manpower = -1
		add_treasury = -50
		p:3754 = {
			set_conquered_by = c:DAT
			establish_city_effect = yes
			while = {
                count = 1
                define_pop = {
					type = citizen
					culture = punici
					religion = canaanitic_pantheon
                }
            }
			while = {
                count = 3
                define_pop = {
					type = freemen
					culture = punici
					religion = canaanitic_pantheon
                }
            }
			while = {
                count = 1
                define_pop = {
					type = slaves
					culture = punici
					religion = canaanitic_pantheon
                }
            }
			if = {
				limit = {
					num_of_port_building < 1
				}
				add_building_level = port_building
			}
			colonize_effect = yes
		}
	}
}
